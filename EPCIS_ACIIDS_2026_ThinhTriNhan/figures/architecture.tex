% DRAW-SPEC (revision 2)
% message: EPCIS events stay on Fabric (L0); per-lot SP1 proofs are folded by a
%          fan-in-8 recursive tree whose ROOT ALONE is wrapped in Groth16 and
%          settled on L1, where it advances an append-only nullifier-set root.
% why redrawn: rev 1 hid the two claims the paper rests on -- the recursion tree
%          (constant on-chain cost) and the nullifier root chain -- and omitted
%          the on-chain PolicyRegistry that enforces condition B1.
% shape encoding (colour is never the only cue):
%   stadium       = actor / organisation   (people, not code)
%   cylinder      = append-only data store (Fabric ledger)
%   plain card    = off-chain computation  (prover programs)
%   double border = EVM smart contract     (on-chain code)
%   small square  = one proof object; diamond = the tree root
% reading order: L0 column top-down, then prover column, then L1 column.
% layout: columns are chained with the positioning library and the zone frames
%         are fitted around them, so no height is hard-coded.
% width: designed for \includegraphics[width=\textwidth] in LNCS (122.5 mm), so
%        the figure is placed at scale 1.0 and all text stays at 7.0 pt.

\documentclass[tikz,border=0pt]{standalone}
\usepackage{draw-paper}
\usetikzlibrary{shapes.geometric,backgrounds,positioning,fit,calc}

\tikzset{
  actor/.style={draw card,rounded corners=3.2mm,fill=drawLight,
                draw=drawInk!45,text width=#1},
  actor/.default=25mm,
  store/.style={cylinder,shape border rotate=90,aspect=0.11,
                draw=drawInk!45,fill=white,line width=0.55pt,align=center,
                inner xsep=2.2mm,inner ysep=1.7mm,text width=#1},
  store/.default=25mm,
  proc/.style={draw card,text width=#1},
  proc/.default=37mm,
  evm/.style={draw card,rounded corners=0.9mm,draw=drawInk!70,
              double,double distance=0.5pt,text width=#1},
  evm/.default=29mm,
  pf/.style={rounded corners=0.35mm,draw=drawInk!55,fill=drawSky!35,
             line width=0.4pt,minimum size=2.7mm,inner sep=0pt},
  root/.style={diamond,draw=drawInk!70,fill=drawAmber!45,line width=0.45pt,
               minimum size=3.4mm,inner sep=0pt,aspect=1.1},
  fan/.style={draw=drawInk!50,line width=0.4pt},
  tag/.style={draw annotation,align=center,text=drawInk,
              font=\sffamily\fontsize{7.0}{7.6}\selectfont},
  badge/.style={circle,draw=drawInk!60,fill=white,line width=0.45pt,
                inner sep=0.3mm,text=drawInk,
                font=\sffamily\bfseries\fontsize{7.0}{7.4}\selectfont},
}
